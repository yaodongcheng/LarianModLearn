KBSECTION
IF
TextEventSet("rcwhtorches")
THEN
GlobalSetFlag("RC_WH_SeaGodStatue_AllTorchesLit");
TimerCancel("RC_WH_SeaGodStatue_AllTorchesLitTimer");
TimerLaunch("RC_WH_SeaGodStatue_AllTorchesLitTimer", 3000);

IF
CharacterEnteredTrigger((CHARACTERGUID)_Var1, TRIGGERGUID_S_RC_WH_SeaGodStatue_DiscoverTrigger_f43c276d-ca61-46e1-a9cb-5663f154e957)
AND
DB_IsPlayer(_Var1)
THEN
ObjectSetFlag(_Var1, "QuestUpdate_RC_WH_SeaGodStatue_DiscoveredStatue");

IF
ItemStatusRemoved((ITEMGUID)_Var1, "BURNING", _)
AND
DB_RC_WH_SeaGodStatue_Torches(_Var1)
THEN
DB_RC_WH_SeaGodStatue_TorchesDoused(_Var1);

IF
ItemStatusChange((ITEMGUID)_Var1, "BURNING", _)
AND
DB_RC_WH_SeaGodStatue_Torches(_Var1)
AND
HasActiveStatus(_Var1, "BURNING", 1)
THEN
NOT DB_RC_WH_SeaGodStatue_TorchesDoused(_Var1);
Proc_RC_WH_SeaGodStatue_CheckAllTorchesLit();
ProcObjectTimerCancel(_Var1, "DouseTimer");
ProcObjectTimer(_Var1, "DouseTimer", 1000);

PROC
ProcObjectTimerFinished((GUIDSTRING)_Var1, "DouseTimer")
AND
DB_RC_WH_SeaGodStatue_Torches(_Var1)
AND
GlobalGetFlag("RC_WH_SeaGodStatue_AllTorchesLit", _Var2)
AND
_Var2 == 0
THEN
RemoveStatus(_Var1, "BURNING");
ProcIncreaseCounter("RC_WH_SeaGodStatue_TorchDoused");
Proc_RC_WH_SeaGodStatue_CheckDouseCounter();

PROC
Proc_RC_WH_SeaGodStatue_CheckAllTorchesLit()
AND NOT
DB_RC_WH_SeaGodStatue_TorchesDoused(_)
THEN
GlobalSetFlag("RC_WH_SeaGodStatue_AllTorchesLit");
TimerCancel("RC_WH_SeaGodStatue_AllTorchesLitTimer");
TimerLaunch("RC_WH_SeaGodStatue_AllTorchesLitTimer", 3000);

IF
TimerFinished("RC_WH_SeaGodStatue_AllTorchesLitTimer")
AND
GlobalGetFlag("RC_WH_SeaGodStatue_RainbowShown", 0)
AND
GetClosestAlivePlayer(ITEMGUID_S_RC_WH_SeaGodStatue_c15f7f1e-b973-4df4-a7c7-5b53bf4544ae, _Var1, _)
THEN
GlobalSetFlag("RC_WH_SeaGodStatue_RainbowShown");

PlaySound(ITEMGUID_S_RC_WH_SeaGodStatue_c15f7f1e-b973-4df4-a7c7-5b53bf4544ae, "SE_RC_WH_SeaGodStatue_Rainbow");

PROC

AND
